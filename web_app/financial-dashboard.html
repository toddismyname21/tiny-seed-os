  • Recovery >= 5%: Resume offensive

Complete Algorithm Pseudocode

MONTHLY_INVESTMENT_DECISION(portfolio_data):

    // PHASE 1: SAFETY CHECKS
    emergency = CHECK_EMERGENCY_RESERVE(reserve, expenses)
    IF emergency == "CRITICAL": RETURN { action: "FUND_RESERVE" }

    // PHASE 2: SEASONAL ADJUSTMENT
    contribution = base * SEASON_MULTIPLIER[current_season]
    // Harvest (Jun-Oct): 1.5x | Planting (Mar-May): 0.75x | Winter: 0.25x

    // PHASE 3: RISK ASSESSMENT
    drawdown = (peak - current) / peak
    IF drawdown >= 0.15: growth_mode = "DEFENSIVE"
    ELSE IF drawdown >= 0.10: growth_mode = "DEFENSIVE"
    ELSE: growth_mode = CALCULATE_MOMENTUM_SIGNAL()

    // PHASE 4: MOMENTUM SIGNAL
    IF growth_mode != "DEFENSIVE":
        FOR EACH asset IN [QQQ, VUG, MTUM, VBK]:
            score[asset] = COMPOSITE_MOMENTUM(1m, 3m, 6m, 12m)
        top_assets = TOP_N(score, n=2)
        passing = FILTER(top_assets, WHERE momentum > SHY_return)
        IF passing.empty: growth_mode = "DEFENSIVE"
        ELSE: growth_holdings = EQUAL_WEIGHT(passing)

    // PHASE 5: BUILD TARGET (75% Safe + 25% Growth)
    target = SAFE_STRATEGY * 0.75 + growth_holdings * 0.25

    // PHASE 6: REBALANCE (use contributions first)
    IF max(drift) >= 0.05 OR new_contribution > 0:
        buy underweight positions with contribution
        IF still drifted: sell overweight, buy underweight

    RETURN { contribution, trades, growth_mode, risk_status }

Signal Calculation Formulas

1. Composite Momentum
composite = 0.25 * mom_1m + 0.25 * mom_3m + 0.25 * mom_6m + 0.25 * mom_12m
2. Absolute Momentum Filter
passes_filter = asset_return_12m > SHY_return_12m
IF passes: INCLUDE in offensive | ELSE: EXCLUDE
3. Rolling Volatility (Annualized)
daily_stddev = sqrt(variance(21_day_returns))
annual_volatility = daily_stddev * sqrt(252)
4. Allocation Drift
drift = current_percent - target_percent
IF abs(drift) >= 0.05: needs_rebalancing = TRUE
